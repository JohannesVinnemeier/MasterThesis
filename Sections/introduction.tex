\section*{Introduction}

\begin{gromov}[\cite{gromov2018metric}]
Let $X$ be a Riemannian manifold, $V \to M$ a two dimensional vector bundle over a compact enlargeable manifold $M$. Assume that 
\begin{enumerate}[label=(\roman*)]
    \item There exists a proper map $f \colon X \to V$, $\mathrm{deg}(f) \neq 0$,
    \item[\textcolor{red}{(ii)}] \textcolor{red}{The total space of the circle bundle corresponding to $V$ is also enlargeable.}
\end{enumerate}
Then 
\[
\min_{B_{x_0}(R)} \mathrm{scal}(X) \leq \frac{8 \pi^2}{(R-R_0)^2}
\]
for some $R_0 = R_0(X,x_0)$ and all $R > R_0$.
\end{gromov}


\begin{question}[\cite{gromov2018metric}]
``What happens to nontrivial bundles $V \to M$? [...] In general, the examples in \cite{BrunnbauerHanke} indicate a possibility of non-enlargeable circle bundles over enlargeable $M$; yet, it seems hard(er) to find such examples, where the corresponding total space would admit metrics with $\mathrm{scal} > 0$.''
\end{question}
